    const corners = polygon.map(([u, v]) => [
      u * s.width * stoneRng.float(0.96, 1.03),
      v * s.depth * stoneRng.float(0.96, 1.03),
    ]);
'''+s[b:]
a=s.index('    const profile =');b=s.index('    const crownCenter',a)

    // interpolate these controls, so refinement cannot move the other stones.
    const control = corners.map(([u, v]) => [u, v, random.float(-0.22, 0.22), random.float(0.92, 1.08)]);
    const cutStart = profileRng.int(0, control.length - 1);
    const wornCorners = !wear ? control : control.flatMap((corner, i) => {
      if ((i + control.length - cutStart) % control.length > 2) return [corner];
      const trim = 0.12 + 0.19 * wear;
      const neighbors = [control[(i + control.length - 1) % control.length], control[(i + 1) % control.length]];
      return neighbors.map(q => corner.map((v, k) => v + (q[k] - v) * trim));
    });
    const shoulder = profileRng.float(0.58, 0.79);
    const profile = wear
      ? [[0, toe], [middle * 0.71, waist], [shoulder, 0.91], [0.92, 0.78], [1, 0.72 - wear * 0.12]]
      : [[0, toe], [middle, waist], [1, crownWidth]];
    for (const [row, [t, scale]] of profile.entries()) {
      const ring: number[] = [];
      for (const [u, v, bend, spread] of wornCorners) {
        const shoulderBend = 0.52 * bend + 0.10 * (u / s.width - v / s.depth);
        const fraction = wear ? t + Math.sin(Math.PI * t) * shoulderBend : row === 1 ? t + bend : t;
        const section = wear ? 1 + Math.sin(Math.PI * t) * (spread - 1) : row === 1 ? spread : 1;
        const p = world(
          u * scale * section + s.rake * fraction,
          v * scale * section + s.lean * fraction,
          s.base + (topHeight(u, v) - s.base) * fraction,
        );
        ring.push(vertex(stone, p, color));
        if (t === 1) top.push(p);
