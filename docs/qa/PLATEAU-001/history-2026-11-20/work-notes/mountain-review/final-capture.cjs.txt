const {chromium}=require('playwright');
const fs=require('node:fs');
const path=require('node:path');
const [port,directory]=process.argv.slice(2);
const views={front:{eye:[-970,191,-395],target:[-6,48,2],fov:17},shoulder:{eye:[-406,147,-224],target:[-165,63,-78],fov:30},rim:{eye:[-328,134,-166],target:[-204,85,-104],fov:34},plateau:{eye:[-156,145,-110],target:[30,103,27],fov:46},route:{eye:[-339,16,-54],target:[-160,50,-27],fov:53},entrance:null};
(async()=>{
 fs.mkdirSync(directory,{recursive:true});
 const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox','--use-gl=angle','--use-angle=swiftshader','--enable-unsafe-swiftshader','--js-flags=--max-old-space-size=6000']});
 try{
 const page=await browser.newPage({viewport:{width:1344,height:610},deviceScaleFactor:1});
 const errors=[];page.on('pageerror',e=>{console.error(e);errors.push(e.message)});page.on('console',m=>{if(m.type()==='error'){console.error(m.text()); errors.push(m.text())}});page.on('response',r=>{if(r.status()>=400) errors.push(r.url()+' '+r.status())});
 await page.goto('http://127.0.0.1:'+port+'/experiments/plateau/mountain.html');
 await page.waitForFunction(()=>window.__mountain?.ready,null,{timeout:420000});
 await page.addStyleTag({content:'aside,#status{display:none!important}'});
 for(const [name,v] of Object.entries(views)){
 await page.evaluate(({name,v})=>{const m=window.__mountain;if(v)m.views[name]=v;m.view(name)}, {name,v});
 await page.locator('canvas').screenshot({path:path.join(directory,name+'.png'), timeout:120000});console.log(name);
 }
 const stats=await page.evaluate(()=>{const m=window.__mountain;return {views:m.views,triangles:m.renderer.info.render.triangles,rockStats:m.dressing.stats,trees:m.groups.trees?.userData.count,snow:m.snow.stats}});
 fs.writeFileSync(path.join(directory,'capture.json'),JSON.stringify({stats,errors},null,2)); if(errors.length)process.exitCode=1;
 }finally{await browser.close()}
})().catch(e=>{console.error(e);process.exitCode=1});
